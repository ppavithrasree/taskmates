+const getMessageReceipt = (message: { senderId: string; deliveredTo?: string[]; readBy?: string[] }, memberIds: string[]): MessageReceipt => {
+  const recipientIds = memberIds.filter((id) => id !== message.senderId);
+  const readBy = new Set([message.senderId, ...normalizeIdArray(message.readBy)]);
+  const deliveredTo = new Set(
+    normalizeIdArray(message.deliveredTo).length > 0 ? normalizeIdArray(message.deliveredTo) : recipientIds
+  );
+  const readCount = recipientIds.filter((id) => readBy.has(id)).length;
+  const deliveredCount = recipientIds.filter((id) => deliveredTo.has(id) || readBy.has(id)).length;
+  const totalCount = recipientIds.length;
+  const allRead = totalCount > 0 && readCount === totalCount;
+  const anyDelivered = deliveredCount > 0;
+  const label = readCount > 0

+    : deliveredCount > 0
+      ? `Delivered to ${deliveredCount}/${totalCount}`
+      : `Delivered to ${totalCount}/${totalCount}`;
